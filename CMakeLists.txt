cmake_minimum_required(VERSION 4.2)
set(CMAKE_TOOLCHAIN_FILE "C:/vcpkg/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "")
set(VCPKG_TARGET_TRIPLET "x64-windows-static" CACHE STRING "")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "")
project(ROCKWheelMenu VERSION 0.1.0 LANGUAGES CXX)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(COMMON_LIB_F4VR_PATH "${WORKSPACE}/libraries_and_tools/CommonLibF4VR" CACHE PATH "")
set(ENABLE_FALLOUT_VR ON CACHE BOOL "" FORCE)
set(ENABLE_FALLOUT_F4 OFF CACHE BOOL "" FORCE)
set(ENABLE_FALLOUT_NG OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
add_subdirectory("${COMMON_LIB_F4VR_PATH}" CommonLibF4 EXCLUDE_FROM_ALL)
find_package(spdlog CONFIG REQUIRED)
set(IMGUI "${WORKSPACE}/../devtools/imgui")
add_library(wheel_imgui STATIC "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp"
  "${IMGUI}/imgui_tables.cpp" "${IMGUI}/imgui_widgets.cpp" "${IMGUI}/backends/imgui_impl_dx11.cpp")
target_include_directories(wheel_imgui PUBLIC "${IMGUI}" "${IMGUI}/backends")
target_compile_features(wheel_imgui PUBLIC cxx_std_23)
target_compile_definitions(wheel_imgui PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(wheel_imgui PUBLIC d3d11 dxgi d3dcompiler)
add_library(wheel_view STATIC src/WheelView.cpp src/Fonts.cpp)
target_include_directories(wheel_view PUBLIC src)
target_link_libraries(wheel_view PUBLIC wheel_imgui)
add_library(ROCKWheelMenu SHARED src/Plugin.cpp src/Runtime.cpp src/Inventory.cpp src/Renderer.cpp src/exports.def)
target_include_directories(ROCKWheelMenu PRIVATE src
 "${WORKSPACE}/main_projects/RPS_SDK/SDK/ROCK/include"
 "${WORKSPACE}/secondary_projects/RPS_UI_Framework/SDK/include")
target_link_libraries(ROCKWheelMenu PRIVATE CommonLibF4::CommonLibF4 spdlog::spdlog wheel_view shell32 version bcrypt)
target_precompile_headers(ROCKWheelMenu PRIVATE src/PCH.h)
target_compile_options(ROCKWheelMenu PRIVATE /MP2 /Zi /wd4715 /utf-8)
target_link_options(ROCKWheelMenu PRIVATE /DEBUG:FULL)
add_executable(ROCKWheelPreview WIN32 preview/Main.cpp "${IMGUI}/backends/imgui_impl_win32.cpp")
target_link_libraries(ROCKWheelPreview PRIVATE wheel_view shell32)
target_compile_options(wheel_view PRIVATE /utf-8)
target_compile_options(ROCKWheelPreview PRIVATE /utf-8)
set(COPY_PLUGIN_BASE_PATH "" CACHE PATH "MO2 mod root for auto-deployment")
if(NOT COPY_PLUGIN_BASE_PATH STREQUAL "")
 add_custom_command(TARGET ROCKWheelMenu POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E make_directory "${COPY_PLUGIN_BASE_PATH}/F4SE/Plugins"
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:ROCKWheelMenu>" "${COPY_PLUGIN_BASE_PATH}/F4SE/Plugins/"
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_PDB_FILE:ROCKWheelMenu>" "${COPY_PLUGIN_BASE_PATH}/F4SE/Plugins/")
endif()
enable_testing()
add_executable(WheelPolicyTests tests/PolicyTests.cpp)
target_include_directories(WheelPolicyTests PRIVATE src)
target_compile_features(WheelPolicyTests PRIVATE cxx_std_23)
add_test(NAME Wheel.Policy COMMAND WheelPolicyTests)
